# Themes and Winamp Skin Import Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the user reskin Greedify's colors, fonts and shapes with built-in themes, hand-written `.greedifytheme` files, and themes generated from classic Winamp `.wsz` skins.

**Architecture:**
- All theme logic lives in `GreedifyKit/Theme/`: the `Theme` model and lenient JSON decoding, the built-in themes, a zip reader, `.wsz` parsing, skin→theme mapping and `ThemeStore`. It's unit-tested with Swift Testing.
- The app turns the selected theme into a `ResolvedTheme` of SwiftUI values, once per change, and puts it in the environment.
- A few themed components (buttons, slider, card, section header, row highlight, text roles) replace the system styles in the main window, the popover and the lyrics panel.

**Tech Stack:** Swift 6, SwiftUI and AppKit (macOS 15), Swift Testing, Apple `Compression` (raw deflate), ImageIO (BMP decoding).

**Spec:** `docs/superpowers/specs/2026-09-30-themes-design.md`

## Global Constraints

- The platform is macOS 15. Package code is Swift 6 language mode. Closures that run on background queues must be `@Sendable` (see CLAUDE.md).
- There are no new third-party dependencies. Zip uses `Compression` and bitmaps use ImageIO.
- Theme files use the extension `.greedifytheme`, contain UTF-8 JSON and have format `version` 1. User themes live in `~/Library/Application Support/Greedify/Themes/`.
- Built-in theme ids are `builtin.system`, `builtin.classic`, `builtin.modern`, `builtin.paper` and `builtin.high-contrast`, in that order. The default selection is `builtin.system`.
- Ranges: `fonts.scale` is 0.8–1.4; the radii are 0–24; `borderWidth` is 0–3.
- Zip limits: archives up to 10 MB, entries up to 5 MB uncompressed. Only `pledit.txt`, `viscolor.txt`, `main.bmp` and `genex.bmp` are read, and entry paths are never used to write files.
- The contrast guard requires text on background ≥ 4.5:1 and `textSecondary` ≥ 3:1. `base` is `dark` when the background's relative luminance is < 0.5.
- The error texts, verbatim:
  - "This looks like a Winamp 3/5 skin; only classic (.wsz) skins are supported."
  - "No color files found in this skin."
  - "Couldn't read this file."
- No Winamp skins are bundled, and built-in themes copy no Winamp or Spotify art.
- The System theme must look exactly like today's app.
- NavigationSplitView columns keep their constant `.fixedMinimum`, which the theme never changes. The player bar height is 76 pt (`regular`) or 64 pt (`compact`).
- The Settings window, the Setup window, context menus and the menu-bar label image stay native.

```swift
import Foundation

/// An sRGB color with alpha, components 0...1. Themes store colors as `#RRGGBB` or `#RRGGBBAA`.
public struct ThemeColor: Hashable, Sendable {
    public var r: Double, g: Double, b: Double, a: Double

    public static let white = ThemeColor(r: 1, g: 1, b: 1)
    public static let black = ThemeColor(r: 0, g: 0, b: 0)

    public init(r: Double, g: Double, b: Double, a: Double = 1) {
        func unit(_ v: Double) -> Double { min(max(v, 0), 1) }
        self.r = unit(r); self.g = unit(g); self.b = unit(b); self.a = unit(a)
    }

    public init(r8: Int, g8: Int, b8: Int) {
        self.init(r: Double(r8) / 255, g: Double(g8) / 255, b: Double(b8) / 255)
    }

    /// `#RRGGBB` or `#RRGGBBAA`, `#` optional, any case, surrounding spaces ignored.
    public init?(hex: String) {
        var s = Substring(hex.trimmingCharacters(in: .whitespaces))
        if s.hasPrefix("#") { s = s.dropFirst() }
        guard s.count == 6 || s.count == 8, s.allSatisfy(\.isHexDigit), let v = UInt64(s, radix: 16) else { return nil }
        func byte(_ shift: UInt64) -> Double { Double((v >> shift) & 0xFF) / 255 }
        if s.count == 6 { self.init(r: byte(16), g: byte(8), b: byte(0)) }
        else { self.init(r: byte(24), g: byte(16), b: byte(8), a: byte(0)) }
    }

    public var hex: String {
        func byte(_ v: Double) -> Int { Int((v * 255).rounded()) }
        let rgb = String(format: "#%02X%02X%02X", byte(r), byte(g), byte(b))
        return byte(a) == 255 ? rgb : rgb + String(format: "%02X", byte(a))
    }

    /// WCAG relative luminance.
    public var luminance: Double {
        func linear(_ c: Double) -> Double { c <= 0.04045 ? c / 12.92 : pow((c + 0.055) / 1.055, 2.4) }
        return 0.2126 * linear(r) + 0.7152 * linear(g) + 0.0722 * linear(b)
    }

    /// WCAG contrast ratio, 1...21.
    public func contrast(with other: ThemeColor) -> Double {
        let (hi, lo) = (max(luminance, other.luminance), min(luminance, other.luminance))
        return (hi + 0.05) / (lo + 0.05)
    }

    /// Moves `amount` (0...1) of the way toward `other`; alpha is kept.
    public func blended(toward other: ThemeColor, _ amount: Double) -> ThemeColor {
        ThemeColor(r: r + (other.r - r) * amount, g: g + (other.g - g) * amount, b: b + (other.b - b) * amount, a: a)
    }

    public func withAlpha(_ alpha: Double) -> ThemeColor { ThemeColor(r: r, g: g, b: b, a: alpha) }

    /// HSL lightness, 0...1.
    public var lightness: Double { (max(r, g, b) + min(r, g, b)) / 2 }

    /// HSL saturation, 0...1.
    public var saturation: Double {
        let hi = max(r, g, b), lo = min(r, g, b)
        guard hi != lo else { return 0 }
        return (hi - lo) / (1 - abs(2 * lightness - 1))
    }
}
```

```swift
import Foundation

public enum ThemeBase: String, Codable, Sendable { case light, dark }
public enum ThemeButtonStyle: String, Codable, Sendable { case flat, outlined, filled, bevel }
public enum ThemeDensity: String, Codable, Sendable { case compact, regular }
public enum ThemeFontWeight: String, Codable, Sendable { case regular, semibold, bold, heavy }
public enum LyricsStyle: String, Codable, Sendable { case artwork, solid }

/// A Greedify theme (spec: docs/superpowers/specs/2026-09-30-themes-design.md §2).
/// Decoding is lenient: every key is optional, and a bad value falls back on its own.
public struct Theme: Equatable, Sendable {
    public static let currentVersion = 1
    public static let scaleRange = 0.8...1.4
    public static let radiusRange = 0.0...24.0
    public static let borderRange = 0.0...3.0

    public struct Colors: Equatable, Sendable {
        public var background, surface, raised, text, textSecondary, accent, onAccent, selection, divider, lyricsText: ThemeColor
    }

    public struct Fonts: Equatable, Sendable {
        public var family: String?
        public var scale: Double
        public var titleWeight: ThemeFontWeight
    }

    public struct Shapes: Equatable, Sendable {
        public var cardRadius, buttonRadius, artworkRadius, rowRadius, borderWidth: Double
        public var buttonStyle: ThemeButtonStyle
        public var density: ThemeDensity
    }

    public var version = Theme.currentVersion
    public var name: String
    public var author: String?
    public var source: String?
    public var base: ThemeBase
    public var colors: Colors
    public var fonts: Fonts
    public var shapes: Shapes
    public var lyrics: LyricsStyle

    /// What a theme gets for anything it leaves out: macOS's own dark or light colors.
    public static func defaults(for base: ThemeBase) -> Theme {
        func c(_ hex: String) -> ThemeColor { ThemeColor(hex: hex)! }
        let colors = base == .dark
            ? Colors(background: c("#1E1E1E"), surface: c("#282828"), raised: c("#323232"), text: c("#F2F2F2"),
                     textSecondary: c("#9A9A9A"), accent: c("#0A84FF"), onAccent: c("#FFFFFF"), selection: c("#3A3A3C"),
                     divider: c("#FFFFFF1A"), lyricsText: c("#FFFFFF"))
            : Colors(background: c("#FFFFFF"), surface: c("#F5F5F7"), raised: c("#FFFFFF"), text: c("#1D1D1F"),
                     textSecondary: c("#6E6E73"), accent: c("#007AFF"), onAccent: c("#FFFFFF"), selection: c("#E5E5EA"),
                     divider: c("#0000001A"), lyricsText: c("#1D1D1F"))
        return Theme(name: "Untitled", author: nil, source: nil, base: base, colors: colors,
                     fonts: Fonts(family: nil, scale: 1, titleWeight: .bold),
                     shapes: Shapes(cardRadius: 8, buttonRadius: 6, artworkRadius: 4, rowRadius: 6, borderWidth: 0,
                                    buttonStyle: .filled, density: .regular),
                     lyrics: .artwork)
    }

    public static func decode(from data: Data) throws -> Theme { try JSONDecoder().decode(Theme.self, from: data) }

    public func encoded() throws -> Data {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        return try encoder.encode(self)
    }
}

extension Theme: Codable {
    private enum Keys: String, CodingKey { case version, name, author, source, base, colors, fonts, shapes, lyrics }
    private enum ColorKeys: String, CodingKey {
        case background, surface, raised, text, textSecondary, accent, onAccent, selection, divider, lyricsText
    }
    private enum FontKeys: String, CodingKey { case family, scale, titleWeight }
    private enum ShapeKeys: String, CodingKey { case cardRadius, buttonRadius, artworkRadius, rowRadius, borderWidth, buttonStyle, density }

    // Computed, not stored: key paths aren't Sendable, so a static let would trip Swift 6's checks.
    private static var colorPaths: [(ColorKeys, WritableKeyPath<Colors, ThemeColor>)] {
        [(.background, \.background), (.surface, \.surface), (.raised, \.raised), (.text, \.text),
         (.textSecondary, \.textSecondary), (.accent, \.accent), (.onAccent, \.onAccent), (.selection, \.selection),
         (.divider, \.divider), (.lyricsText, \.lyricsText)]
    }
    private static var radiusPaths: [(ShapeKeys, WritableKeyPath<Shapes, Double>)] {
        [(.cardRadius, \.cardRadius), (.buttonRadius, \.buttonRadius), (.artworkRadius, \.artworkRadius), (.rowRadius, \.rowRadius)]
    }

    public init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: Keys.self)
        var t = Theme.defaults(for: (try? c.decode(ThemeBase.self, forKey: .base)) ?? .dark)
        t.version = (try? c.decode(Int.self, forKey: .version)) ?? Theme.currentVersion
        if let name = try? c.decode(String.self, forKey: .name), !name.isEmpty { t.name = name }
        t.author = try? c.decode(String.self, forKey: .author)
        t.source = try? c.decode(String.self, forKey: .source)
        if let lyrics = try? c.decode(LyricsStyle.self, forKey: .lyrics) { t.lyrics = lyrics }

        if let cc = try? c.nestedContainer(keyedBy: ColorKeys.self, forKey: .colors) {
            for (key, path) in Self.colorPaths {
                if let hex = try? cc.decode(String.self, forKey: key), let color = ThemeColor(hex: hex) { t.colors[keyPath: path] = color }
            }
        }
        if let fc = try? c.nestedContainer(keyedBy: FontKeys.self, forKey: .fonts) {
            if let family = try? fc.decode(String.self, forKey: .family) {
                let trimmed = family.trimmingCharacters(in: .whitespaces)
                t.fonts.family = trimmed.isEmpty ? nil : trimmed
            }
            if let scale = try? fc.decode(Double.self, forKey: .scale) { t.fonts.scale = scale.clamped(to: Self.scaleRange) }
            if let weight = try? fc.decode(ThemeFontWeight.self, forKey: .titleWeight) { t.fonts.titleWeight = weight }
        }
        if let sc = try? c.nestedContainer(keyedBy: ShapeKeys.self, forKey: .shapes) {
            for (key, path) in Self.radiusPaths {
                if let v = try? sc.decode(Double.self, forKey: key) { t.shapes[keyPath: path] = v.clamped(to: Self.radiusRange) }
            }
            if let v = try? sc.decode(Double.self, forKey: .borderWidth) { t.shapes.borderWidth = v.clamped(to: Self.borderRange) }
            if let v = try? sc.decode(ThemeButtonStyle.self, forKey: .buttonStyle) { t.shapes.buttonStyle = v }
            if let v = try? sc.decode(ThemeDensity.self, forKey: .density) { t.shapes.density = v }
        }
        self = t
    }

    public func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: Keys.self)
        try c.encode(version, forKey: .version)
        try c.encode(name, forKey: .name)
        try c.encodeIfPresent(author, forKey: .author)
        try c.encodeIfPresent(source, forKey: .source)
        try c.encode(base, forKey: .base)
        try c.encode(lyrics, forKey: .lyrics)
        var cc = c.nestedContainer(keyedBy: ColorKeys.self, forKey: .colors)
        for (key, path) in Self.colorPaths { try cc.encode(colors[keyPath: path].hex, forKey: key) }
```

```swift
extension Data {
    mutating func appendLE16(_ v: UInt16) { append(contentsOf: [UInt8(v & 0xFF), UInt8(v >> 8)]) }
    mutating func appendLE32(_ v: UInt32) { appendLE16(UInt16(v & 0xFFFF)); appendLE16(UInt16(v >> 16)) }

/// Builds small zip archives for tests: stored or deflated entries, CRCs left at 0.
struct ZipBuilder {
    private var items: [(path: String, data: Data, deflate: Bool, claimedSize: Int?)] = []

    /// `claimedSize` overrides the uncompressed size written to the headers (for zip-bomb and corrupt-size tests).
    mutating func add(_ path: String, _ data: Data, deflate: Bool = false, claimedSize: Int? = nil) {
        items.append((path, data, deflate && !data.isEmpty, claimedSize))
    }

    func build() -> Data {
        var out = Data(), central = Data()
        for item in items {
            let payload = item.deflate ? Self.deflate(item.data) : item.data
            let name = Data(item.path.utf8)
            let method: UInt16 = item.deflate ? 8 : 0
            let offset = UInt32(out.count)
            let size = UInt32(item.claimedSize ?? item.data.count)
            // Local file header (30 bytes + name).
            out.appendLE32(0x04034b50); out.appendLE16(20); out.appendLE16(0); out.appendLE16(method)
            out.appendLE16(0); out.appendLE16(0); out.appendLE32(0)
            out.appendLE32(UInt32(payload.count)); out.appendLE32(size)
            out.appendLE16(UInt16(name.count)); out.appendLE16(0)
            out.append(name); out.append(payload)
            // Central directory header (46 bytes + name).
            central.appendLE32(0x02014b50); central.appendLE16(20); central.appendLE16(20); central.appendLE16(0)
            central.appendLE16(method); central.appendLE16(0); central.appendLE16(0); central.appendLE32(0)
            central.appendLE32(UInt32(payload.count)); central.appendLE32(size)
            central.appendLE16(UInt16(name.count)); central.appendLE16(0); central.appendLE16(0)
            central.appendLE16(0); central.appendLE16(0); central.appendLE32(0); central.appendLE32(offset)
            central.append(name)
        }
        let centralOffset = UInt32(out.count)
        out.append(central)
        // End of central directory (22 bytes).
        out.appendLE32(0x06054b50); out.appendLE16(0); out.appendLE16(0)
        out.appendLE16(UInt16(items.count)); out.appendLE16(UInt16(items.count))
        out.appendLE32(UInt32(central.count)); out.appendLE32(centralOffset); out.appendLE16(0)
        return out
    }

    static func deflate(_ data: Data) -> Data {
        var out = Data(count: data.count + 1024)
        let written = out.withUnsafeMutableBytes { dst in
            data.withUnsafeBytes { src in
                compression_encode_buffer(dst.bindMemory(to: UInt8.self).baseAddress!, dst.count,
                                          src.bindMemory(to: UInt8.self).baseAddress!, data.count, nil, COMPRESSION_ZLIB)
            }
        }
        return Data(out.prefix(written))
    }
```

```swift
extension Data {
    /// Little-endian reads at a byte offset from the start; callers check bounds first.
    func le16(at offset: Int) -> UInt16 {
        UInt16(self[startIndex + offset]) | UInt16(self[startIndex + offset + 1]) << 8
    }

    func le32(at offset: Int) -> UInt32 {
        UInt32(le16(at: offset)) | UInt32(le16(at: offset + 2)) << 16
```

```swift
import Compression
import Foundation

/// Reads a zip archive in memory (spec §3): `stored` and `deflate` entries only, no encryption, no zip64.
/// CRCs aren't checked; the contents are only parsed for colors. Entry paths are matched, never written to disk.
struct ZipReader {
    static let maxArchiveBytes = 10 * 1_048_576
    static let maxEntryBytes = 5 * 1_048_576

    enum ZipError: Error, Equatable { case tooLarge, corrupt, unsupported }

    struct Entry {
        let path: String
        let method: UInt16
        let compressedSize: Int
        let uncompressedSize: Int
        let localHeaderOffset: Int

        var fileName: String { path.split(whereSeparator: { $0 == "/" || $0 == "\\" }).last.map(String.init) ?? path }
    }

    private let data: Data
    let entries: [Entry]

    init(data: Data) throws {
        guard data.count <= Self.maxArchiveBytes else { throw ZipError.tooLarge }
        self.data = data
        entries = try Self.readCentralDirectory(data)
    }

    /// The entry whose file name (last path component) is `name`, ignoring case.
    func entry(named name: String) -> Entry? {
        entries.first { $0.fileName.caseInsensitiveCompare(name) == .orderedSame }
    }

    func contents(of entry: Entry) throws -> Data {
        guard entry.uncompressedSize <= Self.maxEntryBytes else { throw ZipError.tooLarge }
        let header = entry.localHeaderOffset
        guard header >= 0, header + 30 <= data.count, data.le32(at: header) == 0x04034b50 else { throw ZipError.corrupt }
        let start = header + 30 + Int(data.le16(at: header + 26)) + Int(data.le16(at: header + 28))
        guard start + entry.compressedSize <= data.count else { throw ZipError.corrupt }
        let raw = data.subdata(in: (data.startIndex + start)..<(data.startIndex + start + entry.compressedSize))
        switch entry.method {
        case 0:
            guard raw.count == entry.uncompressedSize else { throw ZipError.corrupt }
            return raw
        case 8:
            return try Self.inflate(raw, size: entry.uncompressedSize)
        default:
            throw ZipError.unsupported
        }
    }

    /// Raw deflate into a buffer one byte larger than claimed, so an entry that inflates past its
    /// declared size is caught instead of being silently truncated.
    private static func inflate(_ source: Data, size: Int) throws -> Data {
        guard size > 0 else { return Data() }
        guard !source.isEmpty else { throw ZipError.corrupt }
        var out = Data(count: size + 1)
        let written = out.withUnsafeMutableBytes { dst in
            source.withUnsafeBytes { src in
                compression_decode_buffer(dst.bindMemory(to: UInt8.self).baseAddress!, size + 1,
                                          src.bindMemory(to: UInt8.self).baseAddress!, source.count, nil, COMPRESSION_ZLIB)
            }
        }
        guard written == size else { throw ZipError.corrupt }
        return Data(out.prefix(size))
    }

    private static func readCentralDirectory(_ d: Data) throws -> [Entry] {
        guard d.count >= 22 else { throw ZipError.corrupt }
        var end: Int?
        var i = d.count - 22
        let floor = max(0, d.count - 22 - 65_535) // the archive comment is at most 64 KB
        while i >= floor {
            if d.le32(at: i) == 0x06054b50 { end = i; break }
            i -= 1
        }
        guard let end else { throw ZipError.corrupt }
        let count = Int(d.le16(at: end + 10))
        var p = Int(d.le32(at: end + 16))
        var entries: [Entry] = []
        for _ in 0..<count {
            guard p + 46 <= d.count, d.le32(at: p) == 0x02014b50 else { throw ZipError.corrupt }
            let flags = d.le16(at: p + 8)
            let method = d.le16(at: p + 10)
            let compressed = Int(d.le32(at: p + 20)), uncompressed = Int(d.le32(at: p + 24))
            let nameLength = Int(d.le16(at: p + 28)), extra = Int(d.le16(at: p + 30)), comment = Int(d.le16(at: p + 32))
            let offset = Int(d.le32(at: p + 42))
            guard p + 46 + nameLength <= d.count else { throw ZipError.corrupt }
            let nameData = d.subdata(in: (d.startIndex + p + 46)..<(d.startIndex + p + 46 + nameLength))
            let path = String(data: nameData, encoding: .utf8) ?? String(data: nameData, encoding: .isoLatin1) ?? ""
            p += 46 + nameLength + extra + comment
            // Skip encrypted entries, folders and macOS AppleDouble metadata.
            if flags & 1 != 0 || path.hasSuffix("/") || path.hasPrefix("__MACOSX/") { continue }
            entries.append(Entry(path: path, method: method, compressedSize: compressed,
                                 uncompressedSize: uncompressed, localHeaderOffset: offset))
        }
        return entries
    }
}
```

```swift
        var d = Data([0x42, 0x4D])
        d.appendLE32(UInt32(54 + imageSize)); d.appendLE32(0); d.appendLE32(54)
        d.appendLE32(40); d.appendLE32(UInt32(width)); d.appendLE32(UInt32(height))
        d.appendLE16(1); d.appendLE16(24); d.appendLE32(0); d.appendLE32(UInt32(imageSize))
        d.appendLE32(2835); d.appendLE32(2835); d.appendLE32(0); d.appendLE32(0)
        for y in (0..<height).reversed() { // BMP rows are stored bottom-up
            for x in 0..<width {
                let (r, g, b) = pixel(x, y)
                d.append(contentsOf: [b, g, r])
            }
            d.append(Data(count: rowSize - width * 3))
        }
        return d
    }
```

```swift
import CoreGraphics
import Foundation
import ImageIO

/// A decoded skin bitmap, kept only while a skin is imported (spec §3).
struct SkinBitmap {
    let width: Int
    let height: Int
    private let rgba: [UInt8]

    init?(data: Data) {
        guard let source = CGImageSourceCreateWithData(data as CFData, nil),
              let image = CGImageSourceCreateImageAtIndex(source, 0, nil),
              image.width > 0, image.height > 0, image.width * image.height <= 4_000_000 else { return nil }
        let (w, h) = (image.width, image.height)
        // Draw into the image's own RGB space so the pixel values aren't color-converted.
        let space = image.colorSpace.flatMap { $0.model == .rgb ? $0 : nil } ?? CGColorSpace(name: CGColorSpace.sRGB)!
        var buffer = [UInt8](repeating: 0, count: w * h * 4)
        let drawn = buffer.withUnsafeMutableBytes { pointer -> Bool in
            guard let context = CGContext(data: pointer.baseAddress, width: w, height: h, bitsPerComponent: 8, bytesPerRow: w * 4,
                                          space: space, bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return false }
            context.draw(image, in: CGRect(x: 0, y: 0, width: w, height: h))
            return true
        }
        guard drawn else { return nil }
        width = w; height = h; rgba = buffer
    }

    /// The pixel at (x, y), y = 0 at the top.
    func color(x: Int, y: Int) -> ThemeColor? {
        guard (0..<width).contains(x), (0..<height).contains(y) else { return nil }
        let i = (y * width + x) * 4
        return ThemeColor(r8: Int(rgba[i]), g8: Int(rgba[i + 1]), b8: Int(rgba[i + 2]))
    }

    /// The average of the most common color bucket (4 bits per channel) across `regions`.
    func dominantColor(in regions: [(xs: Range<Int>, ys: Range<Int>)]) -> ThemeColor? {
        var buckets: [Int: (count: Int, r: Int, g: Int, b: Int)] = [:]
        for region in regions {
            for y in region.ys.clamped(to: 0..<height) {
                for x in region.xs.clamped(to: 0..<width) {
                    let i = (y * width + x) * 4
                    let (r, g, b) = (Int(rgba[i]), Int(rgba[i + 1]), Int(rgba[i + 2]))
                    let key = (r >> 4) << 8 | (g >> 4) << 4 | (b >> 4)
                    let old = buckets[key] ?? (0, 0, 0, 0)
                    buckets[key] = (old.count + 1, old.r + r, old.g + g, old.b + b)
                }
            }
        }
        guard let best = buckets.values.max(by: { $0.count < $1.count }) else { return nil }
        let n = Double(best.count) * 255
        return ThemeColor(r: Double(best.r) / n, g: Double(best.g) / n, b: Double(best.b) / n)
    }
}
```

```swift
import Foundation

/// Why a skin or theme file couldn't be imported; the messages are shown to the user as-is.
public enum ThemeImportError: Error, Equatable, LocalizedError {
    case unsupportedModernSkin, noColorSources, unreadable

    public var errorDescription: String? {
        switch self {
        case .unsupportedModernSkin: "This looks like a Winamp 3/5 skin; only classic (.wsz) skins are supported."
        case .noColorSources: "No color files found in this skin."
        case .unreadable: "Couldn't read this file."
        }
    }
}

/// The color sources of a classic Winamp 2 skin (spec §3). Nothing else in the archive is read.
public struct WinampSkin: Equatable, Sendable {
    public struct PlaylistColors: Equatable, Sendable {
        public var normal: ThemeColor?
        public var current: ThemeColor?
        public var normalBG: ThemeColor?
        public var selectedBG: ThemeColor?
        public var font: String?
    }

    /// Winamp 2.9 `genex.bmp` swatches, read at x = 48, 50, 52… on row 0 (as Webamp does).
    public struct GenexColors: Equatable, Sendable {
        public var itemBackground, itemForeground, windowBackground, buttonText, windowText, divider, selectionBackground: ThemeColor
    }

    public var playlist: PlaylistColors?
    public var visColors: [ThemeColor] = []
    public var frameColor: ThemeColor?
    public var genex: GenexColors?
}

extension WinampSkin {
    public init(archive data: Data) throws {
        let zip: ZipReader
        do { zip = try ZipReader(data: data) } catch { throw ThemeImportError.unreadable }
        if zip.entry(named: "skin.xml") != nil { throw ThemeImportError.unsupportedModernSkin }
        func read(_ name: String) -> Data? { zip.entry(named: name).flatMap { try? zip.contents(of: $0) } }

        self.init()
        playlist = read("pledit.txt").flatMap { Self.parsePledit(Self.text($0)) }
        visColors = read("viscolor.txt").map { Self.parseViscolor(Self.text($0)) } ?? []
        frameColor = read("main.bmp").flatMap(SkinBitmap.init(data:)).flatMap(Self.frameColor(of:))
        genex = read("genex.bmp").flatMap(SkinBitmap.init(data:)).flatMap(Self.genexColors(of:))
        guard playlist != nil || !visColors.isEmpty || frameColor != nil || genex != nil else { throw ThemeImportError.noColorSources }
    }

    /// Skins from the Winamp era are often Windows-1252, not UTF-8.
    static func text(_ data: Data) -> String {
        String(data: data, encoding: .utf8) ?? String(data: data, encoding: .windowsCP1252) ?? String(decoding: data, as: UTF8.self)
    }

    /// The `[Text]` section of `pledit.txt`. Keys ignore case, `#` is optional, trailing comments are dropped.
    static func parsePledit(_ text: String) -> PlaylistColors? {
        var section = "", values: [String: String] = [:]
        for raw in text.components(separatedBy: .newlines) {
            let line = raw.trimmingCharacters(in: .whitespaces)
            if line.hasPrefix("[") { section = line.lowercased(); continue }
            guard section == "[text]", let eq = line.firstIndex(of: "=") else { continue }
            let key = line[..<eq].trimmingCharacters(in: .whitespaces).lowercased()
            values[key] = line[line.index(after: eq)...].trimmingCharacters(in: .whitespaces)
        }
        func color(_ key: String) -> ThemeColor? {
            values[key].flatMap { ThemeColor(hex: String($0.prefix { !$0.isWhitespace && $0 != ";" && $0 != "/" })) }
        }
        let font = values["font"].flatMap { $0.isEmpty ? nil : $0 }
        let colors = PlaylistColors(normal: color("normal"), current: color("current"), normalBG: color("normalbg"),
                                    selectedBG: color("selectedbg"), font: font)
        return colors == PlaylistColors() ? nil : colors
    }

    /// Up to 24 `r,g,b` lines; `//` comments and lines without three 0...255 numbers are skipped.
    static func parseViscolor(_ text: String) -> [ThemeColor] {
        var colors: [ThemeColor] = []
        for line in text.components(separatedBy: .newlines) {
            let code = line.components(separatedBy: "//")[0]
            let numbers = code.split(whereSeparator: { $0 == "," || $0.isWhitespace }).compactMap { Int($0) }
            guard numbers.count >= 3, numbers.prefix(3).allSatisfy({ (0...255).contains($0) }) else { continue }
            colors.append(ThemeColor(r8: numbers[0], g8: numbers[1], b8: numbers[2]))
            if colors.count == 24 { break }
        }
        return colors
    }

    /// `main.bmp`'s dominant color along its top and bottom 14-pixel strips (the window frame).
    static func frameColor(of bitmap: SkinBitmap) -> ThemeColor? {
        let strip = min(14, bitmap.height)
        return bitmap.dominantColor(in: [(0..<bitmap.width, 0..<strip),
                                         (0..<bitmap.width, (bitmap.height - strip)..<bitmap.height)])
    }

    static func genexColors(of bitmap: SkinBitmap) -> GenexColors? {
        let swatches = (0..<7).compactMap { bitmap.color(x: 48 + $0 * 2, y: 0) }
        guard swatches.count == 7 else { return nil }
        return GenexColors(itemBackground: swatches[0], itemForeground: swatches[1], windowBackground: swatches[2],
                           buttonText: swatches[3], windowText: swatches[4], divider: swatches[5], selectionBackground: swatches[6])
    }
}
```

```swift
import Foundation

/// Turns a Winamp skin into a Greedify theme (spec §3).
public enum SkinThemeMapper {
    /// `fontAvailable` says whether a font family is installed; the app passes an NSFontManager check.
    public static func theme(fromArchive data: Data, fileName: String, fontAvailable: (String) -> Bool) throws -> Theme {
        if (fileName as NSString).pathExtension.lowercased() == "wal" { throw ThemeImportError.unsupportedModernSkin }
        return theme(from: try WinampSkin(archive: data), fileName: fileName, fontAvailable: fontAvailable)
    }

    public static func theme(from skin: WinampSkin, fileName: String, fontAvailable: (String) -> Bool) -> Theme {
        let playlist = skin.playlist, genex = skin.genex
        let background = genex?.itemBackground ?? playlist?.normalBG ?? skin.frameColor ?? skin.visColors.first ?? .black
        let base: ThemeBase = background.luminance < 0.5 ? .dark : .light
        let text = ensureContrast(genex?.itemForeground ?? playlist?.normal ?? farthest(from: background), on: background, 4.5)
        let selection = genex?.selectionBackground ?? playlist?.selectedBG ?? background.blended(toward: text, 0.2)

        let shifted = background.blended(toward: base == .dark ? .white : .black, 0.08)
        var surface = skin.frameColor.flatMap { abs($0.lightness - background.lightness) * 100 >= 4 ? $0 : nil } ?? shifted
        if text.contrast(with: surface) < 4.5 { surface = shifted } // the sidebar's text sits on the surface

        var candidates: [ThemeColor] = []
        if let current = playlist?.current { candidates.append(current) }
        if skin.visColors.count > 2 { candidates += skin.visColors[2..<min(18, skin.visColors.count)] } // the spectrum bars
        if let frame = skin.frameColor { candidates.append(frame) }
        let accent = candidates.filter { $0.contrast(with: background) >= 3 }.max { $0.saturation < $1.saturation } ?? text

        var theme = Theme.defaults(for: base)
        theme.name = (fileName as NSString).deletingPathExtension
        theme.source = "Winamp skin \(fileName)"
        theme.colors = Theme.Colors(
            background: background, surface: surface, raised: surface.blended(toward: text, 0.06), text: text,
            textSecondary: ensureContrast(text.blended(toward: background, 0.35), on: background, 3),
            accent: accent, onAccent: accent.contrast(with: .black) >= accent.contrast(with: .white) ? .black : .white,
            selection: selection, divider: text.withAlpha(0.15), lyricsText: text)
        theme.fonts = Theme.Fonts(family: playlist?.font.flatMap { fontAvailable($0) ? $0 : nil }, scale: 1, titleWeight: .bold)
        theme.shapes = Theme.Shapes(cardRadius: 2, buttonRadius: 2, artworkRadius: 2, rowRadius: 2, borderWidth: 1,
                                    buttonStyle: .bevel, density: .compact)
        theme.lyrics = .solid
        return theme
    }

    /// Pushes `color` toward white or black, whichever the background allows more of, until it reaches `ratio`.
    static func ensureContrast(_ color: ThemeColor, on background: ThemeColor, _ ratio: Double) -> ThemeColor {
        guard color.contrast(with: background) < ratio else { return color }
        let target = farthest(from: background)
        for step in 1...20 {
            let candidate = color.blended(toward: target, Double(step) / 20)
            if candidate.contrast(with: background) >= ratio { return candidate }
        }
        return target
    }

    static func farthest(from background: ThemeColor) -> ThemeColor {
        background.contrast(with: .white) >= background.contrast(with: .black) ? .white : .black
    }
}
```

```swift
import Foundation
import Observation

/// Where user themes live. Injected so tests can use an in-memory folder.
public protocol ThemeFolder {
    var url: URL? { get }
    func list() throws -> [String]
    func read(_ name: String) throws -> Data
    func write(_ data: Data, as name: String) throws
    func delete(_ name: String) throws
}

public struct DiskThemeFolder: ThemeFolder {
    public let directory: URL
    public init(directory: URL) { self.directory = directory }
    public var url: URL? { directory }

    public func list() throws -> [String] {
        guard FileManager.default.fileExists(atPath: directory.path) else { return [] }
        return try FileManager.default.contentsOfDirectory(atPath: directory.path)
    }
    public func read(_ name: String) throws -> Data { try Data(contentsOf: directory.appendingPathComponent(name)) }
    public func write(_ data: Data, as name: String) throws {
        try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
        try data.write(to: directory.appendingPathComponent(name), options: .atomic)
    }
    public func delete(_ name: String) throws { try FileManager.default.removeItem(at: directory.appendingPathComponent(name)) }
}

/// One theme in the list: a built-in, a user file, or a user file that failed to load (`theme == nil`).
public struct ThemeEntry: Identifiable, Equatable, Sendable {
    public let id: String
    public let theme: Theme?
    public let error: String?
    public let isBuiltin: Bool

    public init(id: String, theme: Theme?, error: String?, isBuiltin: Bool) {
        self.id = id; self.theme = theme; self.error = error; self.isBuiltin = isBuiltin
    }

    public var isSystem: Bool { id == ThemeStore.systemID }
    public var name: String { theme?.name ?? id }
}

/// Built-in and user themes plus the selection (spec §4).
@MainActor @Observable
public final class ThemeStore {
    public nonisolated static let systemID = "builtin.system"
    static let fileExtension = "greedifytheme"

    public private(set) var entries: [ThemeEntry] = []
    public private(set) var selectedID: String
    @ObservationIgnored private let folder: any ThemeFolder
    @ObservationIgnored private let settings: AppSettings
    @ObservationIgnored private let builtins: [ThemeEntry]

    public init(folder: any ThemeFolder, settings: AppSettings) {
        self.folder = folder
        self.settings = settings
        builtins = Theme.builtins().map { ThemeEntry(id: $0.id, theme: $0.theme, error: nil, isBuiltin: true) }
        selectedID = settings.themeID
        reload()
    }

    public var folderURL: URL? { folder.url }

    /// The selected theme, or System while the selected one is missing or broken (the selection is kept,
    /// so the theme comes back if its file does).
    public var current: ThemeEntry {
        entries.first { $0.id == selectedID && $0.theme != nil }
            ?? builtins.first
            ?? ThemeEntry(id: Self.systemID, theme: Theme.defaults(for: .dark), error: nil, isBuiltin: true)
    }

    public func reload() {
        let names = ((try? folder.list()) ?? [])
            .filter { ($0 as NSString).pathExtension == Self.fileExtension }
            .sorted { $0.localizedStandardCompare($1) == .orderedAscending }
        let user = names.map { name in
            do { return ThemeEntry(id: name, theme: try Theme.decode(from: folder.read(name)), error: nil, isBuiltin: false) }
            catch { return ThemeEntry(id: name, theme: nil, error: "Couldn't read this theme.", isBuiltin: false) }
        }
        entries = builtins + user
    }

    public func select(_ id: String) {
        selectedID = id
        settings.themeID = id
    }

    /// Imports a `.wsz` skin or a `.greedifytheme` file into the Themes folder.
    @discardableResult
    public func importFile(data: Data, fileName: String, fontAvailable: (String) -> Bool) throws -> ThemeEntry {
        let theme: Theme
        switch (fileName as NSString).pathExtension.lowercased() {
        case "wsz", "wal", "zip":
            theme = try SkinThemeMapper.theme(fromArchive: data, fileName: fileName, fontAvailable: fontAvailable)
        case Self.fileExtension:
            do { theme = try Theme.decode(from: data) } catch { throw ThemeImportError.unreadable }
        default:
            throw ThemeImportError.unreadable
        }
        return try save(theme)
    }

    @discardableResult
    public func duplicate(_ id: String) throws -> ThemeEntry {
        guard var theme = entries.first(where: { $0.id == id })?.theme else { throw ThemeImportError.unreadable }
        theme.name += " Copy"
        return try save(theme)
    }

    /// Deletes a user theme; built-ins can't be deleted.
    public func delete(_ id: String) throws {
        guard let entry = entries.first(where: { $0.id == id }), !entry.isBuiltin else { return }
        try folder.delete(id)
        reload()
    }

    private func save(_ theme: Theme) throws -> ThemeEntry {
        var theme = theme
        let names = Set(entries.compactMap(\.theme?.name))
        theme.name = Self.unique(theme.name) { names.contains($0) }
        let ids = Set(entries.map(\.id))
        let file = Self.unique(Self.fileStem(theme.name)) { ids.contains("\($0).\(Self.fileExtension)") } + ".\(Self.fileExtension)"
        try folder.write(theme.encoded(), as: file)
        reload()
        return entries.first { $0.id == file } ?? ThemeEntry(id: file, theme: theme, error: nil, isBuiltin: false)
    }

    /// `base`, or `base 2`, `base 3`… until `taken` says no.
    static func unique(_ base: String, taken: (String) -> Bool) -> String {
        guard taken(base) else { return base }
        var n = 2
        while taken("\(base) \(n)") { n += 1 }
        return "\(base) \(n)"
    }

    /// A safe file name: no path separators, no leading dots.
    static func fileStem(_ name: String) -> String {
        var stem = name.replacingOccurrences(of: "/", with: "-").replacingOccurrences(of: ":", with: "-")
        while stem.hasPrefix(".") { stem.removeFirst() }
        return stem.isEmpty ? "Theme" : stem
    }
}
```

```swift
import AppKit
import GreedifyKit
import SwiftUI

/// The selected theme as SwiftUI values, built once per theme change (spec §5).
/// System uses live system colors, so it looks exactly like the app did before themes.
struct ResolvedTheme: Sendable {
    var isSystem: Bool
    var background, surface, raised, text, textSecondary, textTertiary, accent, onAccent, selection, divider, lyricsText: Color
    var family: String?
    var scale: CGFloat
    var titleWeight: Font.Weight
    var cardRadius, buttonRadius, artworkRadius, rowRadius, borderWidth: CGFloat
    var buttonStyle: ThemeButtonStyle
    var density: ThemeDensity
    var lyrics: LyricsStyle

    init(_ entry: ThemeEntry) {
        let t = entry.theme ?? Theme.defaults(for: .dark)
        isSystem = entry.isSystem
        if entry.isSystem {
            background = Color(nsColor: .windowBackgroundColor)
            surface = Color(nsColor: .underPageBackgroundColor)
            raised = Color(nsColor: .controlBackgroundColor)
            text = .primary
            textSecondary = .secondary
            textTertiary = Color(nsColor: .tertiaryLabelColor)
            accent = .accentColor
            onAccent = .white
            selection = Color.primary.opacity(0.1)
            divider = Color(nsColor: .separatorColor)
            lyricsText = .white
        } else {
            let c = t.colors
            background = Color(c.background); surface = Color(c.surface); raised = Color(c.raised)
            text = Color(c.text); textSecondary = Color(c.textSecondary); textTertiary = Color(c.textSecondary).opacity(0.75)
            accent = Color(c.accent); onAccent = Color(c.onAccent); selection = Color(c.selection)
            divider = Color(c.divider); lyricsText = Color(c.lyricsText)
        }
        family = t.fonts.family // Font.custom falls back to the system font when a family isn't installed
        scale = t.fonts.scale
        titleWeight = switch t.fonts.titleWeight {
        case .regular: .regular
        case .semibold: .semibold
        case .bold: .bold
        case .heavy: .heavy
        }
        cardRadius = t.shapes.cardRadius
        buttonRadius = t.shapes.buttonRadius
        artworkRadius = t.shapes.artworkRadius
        rowRadius = t.shapes.rowRadius
        borderWidth = t.shapes.borderWidth
        buttonStyle = t.shapes.buttonStyle
        density = t.shapes.density
        lyrics = t.lyrics
    }

    static let system = ResolvedTheme(ThemeEntry(id: ThemeStore.systemID, theme: Theme.defaults(for: .dark), error: nil, isBuiltin: true))

    var rowHeight: CGFloat { density == .compact ? 36 : 46 }
    var playerBarHeight: CGFloat { density == .compact ? 64 : 76 }

    func font(_ size: CGFloat, weight: Font.Weight = .regular) -> Font {
        if let family { return Font.custom(family, fixedSize: size * scale).weight(weight) }
        return .system(size: size * scale, weight: weight)
    }

    func font(_ style: ThemeFontStyle) -> Font {
        if isSystem { return style.system }
        switch style {
        case .headline: return font(style.size, weight: .semibold)
        case .title, .display: return font(style.size, weight: titleWeight)
        default: return font(style.size)
        }
    }

    func color(_ role: TextRole) -> Color {
        switch role {
        case .primary: text
        case .secondary: textSecondary
        case .tertiary: textTertiary
        case .accent: accent
        }
    }
}
```

```swift
import GreedifyKit
import SwiftUI

// MARK: Text

extension View {
    func themed(_ role: TextRole) -> some View { modifier(ThemedText(role: role)) }
    func themedFont(_ style: ThemeFontStyle) -> some View { modifier(ThemedFont(style: style)) }
    func themedSurface(_ kind: SurfaceKind) -> some View { modifier(ThemedSurface(kind: kind)) }
    func themedButton(_ role: ThemedButtonRole) -> some View { modifier(ThemedButton(role: role)) }
    /// Hover and selection highlight for sidebar and track rows.
    func themedRow(selected: Bool, hovering: Bool) -> some View { modifier(ThemedRow(selected: selected, hovering: hovering)) }
}

private struct ThemedText: ViewModifier {
    @Environment(\.theme) private var theme
    let role: TextRole
    func body(content: Content) -> some View { content.foregroundStyle(theme.color(role)) }
}

private struct ThemedFont: ViewModifier {
    @Environment(\.theme) private var theme
    let style: ThemeFontStyle
    func body(content: Content) -> some View { content.font(theme.font(style)) }
}

// MARK: Surfaces

enum SurfaceKind { case content, surface, raised }

/// Background, default text color, tint and font for a region. On System only `.surface` changes anything (`.bar`).
private struct ThemedSurface: ViewModifier {
    @Environment(\.theme) private var theme
    let kind: SurfaceKind

    func body(content: Content) -> some View {
        if theme.isSystem {
            if kind == .surface { content.background(.bar) } else { content }
        } else {
            content
                .background { fill.ignoresSafeArea() }
                .foregroundStyle(theme.text)
                .tint(theme.accent)
                .font(theme.font(.body))
        }
    }

    private var fill: Color {
        switch kind {
        case .content: theme.background
        case .surface: theme.surface
        case .raised: theme.raised
        }
    }
}

struct ThemedDivider: View {
    @Environment(\.theme) private var theme
    var body: some View {
        if theme.isSystem { Divider() }
        else { Rectangle().fill(theme.divider).frame(height: max(1, theme.borderWidth)) }
    }
}

// MARK: Rows

private struct ThemedRow: ViewModifier {
    @Environment(\.theme) private var theme
    let selected: Bool
    let hovering: Bool

    func body(content: Content) -> some View {
        content.background(
            RoundedRectangle(cornerRadius: theme.rowRadius)
                .fill(selected ? theme.selection : hovering ? theme.text.opacity(0.05) : .clear)
        )
    }
}

// MARK: Buttons

enum ThemedButtonRole { case primary, secondary }

/// System keeps the native capsule buttons; every other theme draws its own `buttonStyle`.
private struct ThemedButton: ViewModifier {
    @Environment(\.theme) private var theme
    let role: ThemedButtonRole

    func body(content: Content) -> some View {
        if theme.isSystem {
            if role == .primary { content.buttonStyle(.borderedProminent).buttonBorderShape(.capsule) }
            else { content.buttonStyle(.bordered).buttonBorderShape(.capsule) }
        } else {
            content.buttonStyle(ThemedButtonStyle(role: role, theme: theme))
        }
    }
}

struct ThemedButtonStyle: ButtonStyle {
    let role: ThemedButtonRole
    let theme: ResolvedTheme

    func makeBody(configuration: Configuration) -> some View {
        let pressed = configuration.isPressed
        let shape = RoundedRectangle(cornerRadius: theme.buttonRadius)
        let primary = role == .primary
        configuration.label
            .font(theme.font(13, weight: .semibold))
            .padding(.horizontal, 14)
            .padding(.vertical, theme.density == .compact ? 4 : 6)
            .foregroundStyle(foreground(primary))
            .background { shape.fill(fill(primary, pressed: pressed)) }
            .overlay {
                switch theme.buttonStyle {
                case .outlined: shape.strokeBorder(primary ? theme.accent : theme.divider, lineWidth: max(1, theme.borderWidth))
                case .bevel: BevelEdges(pressed: pressed, width: max(1, theme.borderWidth))
                case .flat, .filled: EmptyView()
                }
            }
            .offset(x: theme.buttonStyle == .bevel && pressed ? 1 : 0, y: theme.buttonStyle == .bevel && pressed ? 1 : 0)

    private func foreground(_ primary: Bool) -> Color {
        switch theme.buttonStyle {
        case .flat, .outlined: primary ? theme.accent : theme.text
        case .filled, .bevel: primary ? theme.onAccent : theme.text
        }
    }

    private func fill(_ primary: Bool, pressed: Bool) -> Color {
        switch theme.buttonStyle {
        case .flat: .clear
        case .outlined: pressed ? theme.selection : .clear
        case .filled, .bevel: (primary ? theme.accent : theme.raised).opacity(pressed && theme.buttonStyle == .filled ? 0.8 : 1)
        }
    }

/// Winamp-style 3D edge: light top-left, dark bottom-right, swapped while pressed.
struct BevelEdges: View {
    let pressed: Bool
    let width: CGFloat

    var body: some View {
        Canvas { context, size in
            let light = Color.white.opacity(0.55), dark = Color.black.opacity(0.55)
            var topLeft = Path()
            topLeft.move(to: CGPoint(x: 0, y: size.height))
            topLeft.addLine(to: .zero)
            topLeft.addLine(to: CGPoint(x: size.width, y: 0))
            var bottomRight = Path()
            bottomRight.move(to: CGPoint(x: size.width, y: 0))
            bottomRight.addLine(to: CGPoint(x: size.width, y: size.height))
            bottomRight.addLine(to: CGPoint(x: 0, y: size.height))
            // Strokes are centred on the edge and the canvas clips the outer half, so 2× width shows `width`.
            context.stroke(topLeft, with: .color(pressed ? dark : light), lineWidth: width * 2)
            context.stroke(bottomRight, with: .color(pressed ? light : dark), lineWidth: width * 2)
        }
        .allowsHitTesting(false)
    }
}

// MARK: Slider

/// A slider drawn from the theme; System keeps the native mini `Slider`.
struct ThemedSlider: View {
    @Environment(\.theme) private var theme
    @Binding var value: Double
    let range: ClosedRange<Double>
    var onEditingChanged: (Bool) -> Void = { _ in }
    @State private var dragging = false

    var body: some View {
        if theme.isSystem {
            Slider(value: $value, in: range, onEditingChanged: onEditingChanged).controlSize(.mini)
        } else {
            GeometryReader { geo in
                let span = range.upperBound - range.lowerBound
                let fraction = span > 0 ? min(max((value - range.lowerBound) / span, 0), 1) : 0
                let knob: CGFloat = 10
                let x = CGFloat(fraction) * max(0, geo.size.width - knob)
                let bevel = theme.buttonStyle == .bevel
                ZStack(alignment: .leading) {
                    RoundedRectangle(cornerRadius: bevel ? 0 : 2).fill(theme.divider).frame(height: 4)
                    RoundedRectangle(cornerRadius: bevel ? 0 : 2).fill(theme.accent).frame(width: x + knob / 2, height: 4)
                    Group {
                        if bevel {
                            Rectangle().fill(theme.raised).overlay(BevelEdges(pressed: dragging, width: 1)).frame(width: knob, height: 12)
                        } else {
                            Circle().fill(theme.text).frame(width: knob, height: knob)
                        }
                    }
                    .offset(x: x)
                }
                .frame(maxHeight: .infinity)
                .contentShape(Rectangle())
                .gesture(
                    DragGesture(minimumDistance: 0)
                        .onChanged { drag in
                            if !dragging { dragging = true; onEditingChanged(true) }
                            let f = min(max((drag.location.x - knob / 2) / max(1, geo.size.width - knob), 0), 1)
                            value = range.lowerBound + Double(f) * span
                        }
                        .onEnded { _ in dragging = false; onEditingChanged(false) }
                )
            }
            .frame(height: 14)
        }
    }
}
```

```swift
import AppKit
import GreedifyKit
import SwiftUI

/// Settings → Themes (spec §6). The Settings window itself stays native.
struct ThemeSettings: View {
    @Environment(AppModel.self) private var model
    @State private var dropTargeted = false

    var body: some View {
        let store = model.themes
        VStack(alignment: .leading, spacing: 12) {
            ScrollView {
                LazyVGrid(columns: [GridItem(.adaptive(minimum: 120), spacing: 14)], spacing: 14) {
                    ForEach(store.entries) { entry in
                        ThemeSwatch(entry: entry, selected: entry.id == store.current.id)
                            .onTapGesture { if entry.theme != nil { model.selectTheme(entry.id) } }
                            .contextMenu { menu(for: entry) }
                    }
                }
                .padding(4)
            }
            .frame(height: 290)
            .overlay {
                if dropTargeted { RoundedRectangle(cornerRadius: 8).strokeBorder(Color.accentColor, lineWidth: 2) }
            }
            .dropDestination(for: URL.self) { urls, _ in
                urls.forEach(model.importTheme(from:))
                return !urls.isEmpty
            } isTargeted: { dropTargeted = $0 }
            HStack {
                Button("Import Skin or Theme…", action: pickFiles)
                Button("Show Themes Folder") { model.showThemesFolder() }
                Spacer()
                Link("Browse the Winamp Skin Museum", destination: URL(string: "https://skins.webamp.org")!)
            }
        }
        .padding(20)
        .modifier(BannerOverlay())
        .onAppear { model.reloadThemes() }
    }

    @ViewBuilder func menu(for entry: ThemeEntry) -> some View {
        if !entry.isBuiltin { Button("Show in Finder") { model.revealTheme(entry) } }
        Button("Duplicate") { model.duplicateTheme(entry) }.disabled(entry.theme == nil)
        if !entry.isBuiltin {
            Divider()
            Button("Delete", role: .destructive) { model.deleteTheme(entry) }
        }
    }

    func pickFiles() {
        let panel = NSOpenPanel()
        panel.allowsMultipleSelection = true
        panel.canChooseDirectories = false
        panel.allowedContentTypes = ["wsz", "wal", "greedifytheme"].compactMap { UTType(filenameExtension: $0) }
        guard panel.runModal() == .OK else { return }
        panel.urls.forEach(model.importTheme(from:))
    }
}

/// A 120×80 preview of a theme: surface strip, background, title text, accent pill and a line of secondary text.
struct ThemeSwatch: View {
    let entry: ThemeEntry
    let selected: Bool

    var body: some View {
        VStack(spacing: 6) {
            ZStack(alignment: .topTrailing) {
                preview.frame(width: 120, height: 80)
                if selected {
                    Image(systemName: "checkmark.circle.fill")
                        .symbolRenderingMode(.palette)
                        .foregroundStyle(.white, Color.accentColor)
                        .padding(5)
                }
            }
            Text(entry.name).font(.caption).lineLimit(1)
        }
        .help(entry.error ?? entry.theme?.source ?? entry.name)
        .opacity(entry.theme == nil ? 0.5 : 1)
    }

    @ViewBuilder var preview: some View {
        let shape = RoundedRectangle(cornerRadius: 8)
        if entry.theme == nil {
            shape.fill(.quaternary).overlay(Image(systemName: "exclamationmark.triangle").foregroundStyle(.secondary))
        } else {
            let t = ResolvedTheme(entry)
            HStack(spacing: 0) {
                t.surface.frame(width: 30)
                VStack(alignment: .leading, spacing: 6) {
                    Text("Aa").font(t.font(15, weight: t.titleWeight)).foregroundStyle(t.text)
                    RoundedRectangle(cornerRadius: min(t.buttonRadius, 6)).fill(t.accent).frame(width: 44, height: 12)
                    RoundedRectangle(cornerRadius: 2).fill(t.textSecondary).frame(width: 60, height: 4)
                }
                .padding(8)
                .frame(maxWidth: .infinity, maxHeight: .infinity, alignment: .topLeading)
                .background(t.background)
            }
            .clipShape(shape)
            .overlay(shape.strokeBorder(selected ? Color.accentColor : Color.primary.opacity(0.15), lineWidth: selected ? 2 : 1))
        }
    }
}
```
